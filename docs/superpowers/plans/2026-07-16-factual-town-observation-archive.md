# Factual Town Observation Archive Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an immutable, evidence-linked daily archive that records the whole town and exports a self-contained Markdown report plus CSV/JSON files without generating research conclusions.

**Architecture:** Convex stores append-only observation events, a small cursor per resident for world-state transition detection, and versioned daily report records. Pure TypeScript modules assemble deterministic day snapshots and factual Markdown/CSV/JSON output; the local model may only compress cited text and is never a fact source. React provides date selection, report preview, evidence drill-down, and a ZIP download while the simulation continues running.

**Tech Stack:** TypeScript 5, Convex, React 18, Jest/ts-jest, `fflate` for browser ZIP generation, existing Pixi/AI Town runtime, local Gemma 12B only for optional text compression.

```ts
const SHANGHAI_OFFSET_MS = 8 * 60 * 60 * 1000;
export function shanghaiDay(timestamp: number) {

  const start = Date.parse(`${day}T00:00:00.000Z`) - SHANGHAI_OFFSET_MS;
```

```ts
export function normalizeObservation(input: {
  sourceType: string; sourceId: string; occurredAt: number; kind: ObservationKind;
  residentId?: string; otherResidentIds?: string[]; landmarkId?: string;
  position?: { x: number; y: number }; observerIntervention?: boolean;
  text: string; payload?: unknown; quality?: string;
}) {
  return {
    ...input,
    day: shanghaiDay(input.occurredAt),
    recordedAt: Date.now(),
    evidenceId: evidenceId(input.sourceType, input.sourceId),
    idempotencyKey: `${input.sourceType}:${input.sourceId}`,
    otherResidentIds: input.otherResidentIds ?? [],
    observerIntervention: input.observerIntervention ?? false,
    payload: input.payload ?? {},
    quality: input.quality ?? 'complete',
  };
}

export async function appendObservation(ctx: MutationCtx, worldId: Id<'worlds'>, input: ObservationInput) {
  const event = normalizeObservation(input);
  const existing = await ctx.db.query('observationEvents').withIndex('idempotency', (q) =>
    q.eq('worldId', worldId).eq('idempotencyKey', event.idempotencyKey),
  ).unique();
  if (existing) return existing._id;
  return await ctx.db.insert('observationEvents', { ...event, worldId });
}
```

```ts
export function assembleDailySnapshot(input: SnapshotInput): DailyObservationSnapshot {
  const timeline = [...input.events].sort((a, b) =>
    a.occurredAt - b.occurredAt || a.evidenceId.localeCompare(b.evidenceId),
  );
  return {
    day: input.day,
    generatedAt: input.generatedAt,
    sealed: input.sealed,
    metadata: buildMetadata(input, timeline),
    timeline,
    residents: buildResidentDayRows(input, timeline),
    relationships: buildRelationshipRows(input, timeline),
    placeVisits: pairArrivalsAndDepartures(timeline),
    institutionUse: buildInstitutionRows(timeline),
    economicEvents: timeline.filter(isEconomicEvent),
    conversations: buildConversationRows(input.messages, timeline),
    publicEvents: timeline.filter(isPublicEvent),
    cultureRecords: collectExplicitCultureRecords(timeline),
    dataQuality: buildQualitySummary(timeline, input),
  };
}
```

Implement the referenced helpers in the same file with these deterministic rules:

```ts
const buildMetadata = (input: SnapshotInput, timeline: ObservationEvent[]) => ({
  residentCount: input.residents.length,
  eventCount: timeline.length,
  messageCount: input.messages.length,
  conversationCount: input.conversations.length,
  observerInterventionCount: timeline.filter((event) => event.observerIntervention).length,
});
const buildResidentDayRows = (input: SnapshotInput, timeline: ObservationEvent[]) =>
  input.residents.map((resident) => ({
    ...resident,
    eventCount: timeline.filter((event) => event.residentId === resident.id).length,
    events: timeline.filter((event) => event.residentId === resident.id),
  }));
const buildRelationshipRows = (input: SnapshotInput, timeline: ObservationEvent[]) => ({
  baseline: input.relationships,
  events: timeline.filter((event) =>
    event.kind === 'relationship_signal' || event.kind === 'relationship_changed'),
});
const pairArrivalsAndDepartures = (timeline: ObservationEvent[]) => {
  const open = new Map<string, ObservationEvent>();
  const visits: Array<Record<string, unknown>> = [];
  for (const event of timeline) {
    const key = `${event.residentId}:${event.landmarkId}`;
    if (event.kind === 'place_arrived') open.set(key, event);
    if (event.kind === 'place_departed' && open.has(key)) {
      const arrived = open.get(key)!;
      visits.push({ residentId: event.residentId, landmarkId: event.landmarkId,
        landmarkName: (arrived.payload as { landmarkName?: string }).landmarkName,
        arrivedAt: arrived.occurredAt, departedAt: event.occurredAt, status: 'completed',
        evidenceIds: [arrived.evidenceId, event.evidenceId] });
      open.delete(key);
    }
  }
  for (const arrived of open.values()) visits.push({ residentId: arrived.residentId,
    landmarkId: arrived.landmarkId,
    landmarkName: (arrived.payload as { landmarkName?: string }).landmarkName,
    arrivedAt: arrived.occurredAt, status: 'open',
    evidenceIds: [arrived.evidenceId] });
  return visits;
};
const groupBy = <T>(rows: T[], key: (row: T) => string) => rows.reduce<Record<string, T[]>>(
  (groups, row) => ((groups[key(row)] ??= []).push(row), groups), {});
const buildInstitutionRows = (timeline: ObservationEvent[]) =>
  Object.entries(groupBy(timeline.filter((event) => event.kind === 'service_used'),
    (event) => event.landmarkId ?? 'unknown')).map(([landmarkId, events]) => ({ landmarkId, events }));
const buildConversationRows = (messages: SnapshotInput['messages'], timeline: ObservationEvent[]) =>
  Object.entries(groupBy(messages, (message) => message.conversationId)).map(([id, rows]) => ({
    id, messages: rows, evidenceIds: timeline.filter((event) =>
      event.payload && (event.payload as { conversationId?: string }).conversationId === id)
      .map((event) => event.evidenceId),
  }));
const collectExplicitCultureRecords = (timeline: ObservationEvent[]) =>
  timeline.filter((event) => (event.payload as { tags?: string[] })?.tags?.includes('culture'));
const buildQualitySummary = (_timeline: ObservationEvent[], input: SnapshotInput) => ({
  gaps: input.serviceGaps, complete: input.serviceGaps.length === 0,
});
```

```ts
export const csvCell = (value: unknown) => {
```

```ts
export function buildObservationArchive(input: ObservationExport) {
  const files: Record<string, Uint8Array> = {};

  files['manifest.json'] = strToU8(JSON.stringify(input.manifest, null, 2));
  return zipSync(files, { level: 6 });
}
```
